/* the button is a tiny state
   machine; the label swaps while
   the icon gets its own motion */
.publish.is-loading .cloud {
  opacity: 0;
  transform: translateY(-8px);
}
.publish.is-loading .spinner {
  opacity: 1;
  animation: spin .7s linear infinite;
}
.publish.is-done .cloud,
.publish.is-done .spinner {
  opacity: 0;
}
.publish.is-done .check {
  opacity: 1;
  transform: scale(1);
}
@keyframes spin {

app.js
// publish → loading → done
// each button owns its own state
const buttons = document
  .querySelectorAll('[data-publish]');

buttons.forEach(button => {
  button.addEventListener('click', () => {
    if (button.dataset.state ===
        'loading') return;

    button.dataset.state = 'loading';
    button.classList.add('is-loading');
    button.querySelector('.label')
      .textContent = 'Loading';

    setTimeout(() => {
      button.dataset.state = 'done';
      button.classList.remove('is-loading');
      button.classList.add('is-done');
      button.querySelector('.label')
        .textContent = 'Done';
    }, 1450);
  });
});
